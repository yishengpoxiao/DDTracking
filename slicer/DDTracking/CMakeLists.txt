set(MODULE_NAME DDTracking)

slicerMacroBuildScriptedModule(
  NAME ${MODULE_NAME}
  SCRIPTS ${MODULE_NAME}.py
)

set(DDTRACKING_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(DDTRACKING_BUILD_ROOT "${CMAKE_BINARY_DIR}/${Slicer_QTSCRIPTEDMODULES_LIB_DIR}")
foreach(DDTRACKING_SUPPORT_DIR slicer_backend model tracker utils)
  file(COPY "${DDTRACKING_ROOT}/${DDTRACKING_SUPPORT_DIR}"
    DESTINATION "${DDTRACKING_BUILD_ROOT}"
    FILES_MATCHING PATTERN "*.py" PATTERN "__pycache__" EXCLUDE)
  install(DIRECTORY "${DDTRACKING_ROOT}/${DDTRACKING_SUPPORT_DIR}"
    DESTINATION "${Slicer_INSTALL_QTSCRIPTEDMODULES_LIB_DIR}"
    PATTERN "__pycache__" EXCLUDE)
endforeach()
file(MAKE_DIRECTORY "${DDTRACKING_BUILD_ROOT}/resources")
file(COPY "${DDTRACKING_ROOT}/resources/MNI_FA_template.nii.gz"
  DESTINATION "${DDTRACKING_BUILD_ROOT}/resources")
install(FILES "${DDTRACKING_ROOT}/resources/MNI_FA_template.nii.gz"
  DESTINATION "${Slicer_INSTALL_QTSCRIPTEDMODULES_LIB_DIR}/resources")
